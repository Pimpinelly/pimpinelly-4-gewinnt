<script>

const ROWS = 6;

const COLUMNS = 7;

let board = Array.from({ length: ROWS }, () => Array(COLUMNS).fill(0));

let gameActive = true;


function createBoard() {

const boardElement = document.getElementById('board');

boardElement.innerHTML = '';

for (let r = 0; r < ROWS; r++) {

for (let c = 0; c < COLUMNS; c++) {

const cell = document.createElement('div');

cell.classList.add('cell');

cell.dataset.row = r;

cell.dataset.col = c;

cell.onclick = () => playerMove(c);

boardElement.appendChild(cell);

}

}

}


function playerMove(c) {

if (!gameActive) return;

let r = getLowestEmptyRow(c);

if (r !== -1) {

placeStone(r, c, 1);

if (checkWin(r, c)) {

document.getElementById('status').innerText = "Pimpinelly gewinnt! 🤡";

new Audio('tusch.mp3').play();

gameActive = false;

} else {

document.getElementById('status').innerText = "Computer denkt nach...";

setTimeout(makeMove, 500);

}

}

}


function makeMove() {

if (!gameActive) return;

let bestScore = -Infinity;

let move = -1;


for (let c = 0; c < COLUMNS; c++) {

let r = getLowestEmptyRow(c);

if (r !== -1) {

let score = 0;

board[r][c] = 2; // Teste Sieg für Blau

if (checkWin(r, c)) score = 1000;

else {

board[r][c] = 1; // Teste Block für Orange

if (checkWin(r, c)) score = 100;

else if (c === 3) score = 3; // Mitte bevorzugen

}

board[r][c] = 0;


if (score > bestScore) {

bestScore = score;

move = c;

}

}

}


if (move === -1) {

let available = [];

for(let c=0; c<COLUMNS; c++) if(board[0][c] === 0) available.push(c);

move = available[Math.floor(Math.random()*available.length)];

}


let r = getLowestEmptyRow(move);

placeStone(r, move, 2);

if (checkWin(r, move)) {

document.getElementById('status').innerText = "Computer gewinnt!";

gameActive = false;

} else {

document.getElementById('status').innerText = "Du bist dran!";

}

}


function getLowestEmptyRow(c) {

for (let r = ROWS - 1; r >= 0; r--) {

if (board[r][c] === 0) return r;

}

return -1;

}


function placeStone(r, c, p) {

board[r][c] = p;

const cell = document.querySelector(`[data-row='${r}'][data-col='${c}']`);

cell.classList.add(p === 1 ? 'player' : 'computer');

}


function checkWin(r, c) {

const p = board[r][c];

const directions = [[0,1], [1,0], [1,1], [1,-1]];

for (let [dr, dc] of directions) {

let count = 1;

for (let i = 1; i < 4; i++) {

let nr = r + dr*i, nc = c + dc*i;

if (nr >= 0 && nr < ROWS && nc >= 0 && nc < COLUMNS && board[nr][nc] === p) count++;

else break;

}

for (let i = 1; i < 4; i++) {

let nr = r - dr*i, nc = c - dc*i;

if (nr >= 0 && nr < ROWS && nc >= 0 && nc < COLUMNS && board[nr][nc] === p) count++;

else break;

}

if (count >= 4) return true;

}

return false;

}


function resetGame() {

board = Array.from({ length: ROWS }, () => Array(COLUMNS).fill(0));

gameActive = true;

document.getElementById('status').innerText = "Du bist dran!";

createBoard();

}


createBoard();

</script>
