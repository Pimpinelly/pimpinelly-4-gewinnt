<script>

const ROWS = 6;

const COLS = 7;

let board = [];

let gameOver = false;

const boardElement = document.getElementById('game-board');

const statusElement = document.getElementById('status');

const tusch = document.getElementById('tuschSound');


function initBoard() {

boardElement.innerHTML = '';

board = Array(ROWS).fill().map(() => Array(COLS).fill(null));

for (let r = 0; r < ROWS; r++) {

for (let c = 0; c < COLS; c++) {

const cell = document.createElement('div');

cell.classList.add('cell');

cell.dataset.row = r;

cell.dataset.col = c;

cell.onclick = () => playerMove(c);

boardElement.appendChild(cell);

}

}

}


function playerMove(col) {

if (gameOver) return;

if (makeMove(col, 'orange')) {

if (checkWin('orange')) {

statusElement.innerText = "Gewonnen! Ein Hoch auf Pimpinelly!";

playSound();

gameOver = true;

} else {

statusElement.innerText = "Handy überlegt...";

setTimeout(aiMove, 600);

}

}

}


function aiMove() {

if (gameOver) return;

let availableCols = [];

for(let c=0; c<COLS; c++) if(board[0][c] === null) availableCols.push(c);

if(availableCols.length > 0) {

let col = availableCols[Math.floor(Math.random() * availableCols.length)];

makeMove(col, 'blue');

if (checkWin('blue')) {

statusElement.innerText = "Das Handy hat gewonnen!";

playSound();

gameOver = true;

} else {

statusElement.innerText = "Du bist dran (Orange)";

}

}

}


function makeMove(col, color) {

for (let r = ROWS - 1; r >= 0; r--) {

if (!board[r][col]) {

board[r][col] = color;

updateUI();

return true;

}

}

return false;

}


function updateUI() {

const cells = document.querySelectorAll('.cell');

cells.forEach(cell => {

const r = cell.dataset.row;

const c = cell.dataset.col;

cell.className = 'cell' + (board[r][c] ? ' ' + board[r][c] : '');

});

}


function playSound() {

tusch.currentTime = 0;

tusch.play().catch(e => console.log("Audio-Hinweis: Interaktion nötig für Sound."));

}


function checkWin(player) {

for (let r = 0; r < ROWS; r++) {

for (let c = 0; c < COLS; c++) {

if (checkDirection(r, c, 0, 1, player) ||

checkDirection(r, c, 1, 0, player) ||

checkDirection(r, c, 1, 1, player) ||

checkDirection(r, c, 1, -1, player)) return true;

}

}

return false;

}


function checkDirection(r, c, dr, dc, p) {

let count = 0;

for (let i = 0; i < 4; i++) {

let nr = r + i * dr;

let nc = c + i * dc;

if (nr >= 0 && nr < ROWS && nc >= 0 && nc < COLS && board[nr][nc] === p) count++;

else break;

}

return count === 4;

}


function resetGame() {

gameOver = false;

statusElement.innerText = "Du bist dran (Orange)";

initBoard();

}


initBoard();
